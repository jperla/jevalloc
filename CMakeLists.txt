cmake_minimum_required(VERSION 3.19)
project(jevalloc VERSION 0.1 LANGUAGES C CXX)
set(CMAKE_C_STANDARD 11)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_BUILD_TYPE Release CACHE STRING "Build type")
add_compile_options(-O3 -g -fno-omit-frame-pointer)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  add_compile_options(-march=armv8.1-a)
endif()
find_package(Threads REQUIRED)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
set(MI ${CMAKE_CURRENT_SOURCE_DIR}/third_party/mimalloc)
set(MI_SOURCES alloc alloc-aligned alloc-posix arena bitmap heap init libc options os page random segment segment-map stats prim/prim)
set(MI_FILES)
foreach(s IN LISTS MI_SOURCES)
 list(APPEND MI_FILES ${MI}/src/${s}.c)
endforeach()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/patch_mimalloc.py ${MI} ${CMAKE_CURRENT_BINARY_DIR}/mi-patched ${CMAKE_CURRENT_SOURCE_DIR}/patches/mimalloc-class-tags.patch COMMAND_ERROR_IS_FATAL ANY)
set(MI_CORE_FILES ${MI_FILES})
list(REMOVE_ITEM MI_CORE_FILES ${MI}/src/segment.c ${MI}/src/heap.c ${MI}/src/init.c ${MI}/src/alloc.c)
list(APPEND MI_CORE_FILES ${CMAKE_CURRENT_BINARY_DIR}/mi-patched/src/segment.c ${CMAKE_CURRENT_BINARY_DIR}/mi-patched/src/heap.c ${CMAKE_CURRENT_BINARY_DIR}/mi-patched/src/init.c ${CMAKE_CURRENT_BINARY_DIR}/mi-patched/src/alloc.c)
add_library(mi-core STATIC ${MI_CORE_FILES})
set_target_properties(mi-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
target_include_directories(mi-core PUBLIC ${MI}/include)
target_include_directories(mi-core PRIVATE ${MI}/src)
target_compile_definitions(mi-core PRIVATE MI_STATIC_LIB MI_BUILD_RELEASE)
if(APPLE)
 list(APPEND MI_FILES ${MI}/src/prim/osx/alloc-override-zone.c)
endif()
add_library(mimalloc SHARED ${MI_FILES})
target_include_directories(mimalloc PRIVATE ${MI}/include)
target_compile_definitions(mimalloc PRIVATE MI_SHARED_LIB MI_SHARED_LIB_EXPORT MI_BUILD_RELEASE MI_MALLOC_OVERRIDE)
if(APPLE)
  target_compile_definitions(mi-core PRIVATE MI_TLS_SLOT=89)
  target_compile_definitions(mimalloc PRIVATE MI_OSX_ZONE=1 MI_OSX_INTERPOSE=1)
endif()
execute_process(COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tools/override.py ${MI} ${CMAKE_CURRENT_BINARY_DIR} COMMAND_ERROR_IS_FATAL ANY)
add_library(jevalloc SHARED src/jevalloc.c src/new.cpp ${CMAKE_CURRENT_BINARY_DIR}/override.c ${CMAKE_CURRENT_BINARY_DIR}/zone.c)
if(NOT APPLE)
  set_target_properties(jevalloc PROPERTIES LINKER_LANGUAGE C)
  set(CMAKE_CXX_IMPLICIT_LINK_LIBRARIES "")
  # Equal optimization for the baseline and shim; allow local-call optimization.
  foreach(target mi-core mimalloc jevalloc)
    target_compile_options(${target} PRIVATE -fno-semantic-interposition)
    set_property(TARGET ${target} PROPERTY INTERPROCEDURAL_OPTIMIZATION TRUE)
  endforeach()
endif()
target_include_directories(jevalloc PRIVATE src ${MI}/include)
target_link_libraries(jevalloc PRIVATE mi-core Threads::Threads ${CMAKE_DL_LIBS})
target_link_libraries(mimalloc PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
add_executable(stress tests/stress.cpp)
target_compile_options(stress PRIVATE -UNDEBUG -fno-builtin -fsized-deallocation)
target_link_libraries(stress PRIVATE Threads::Threads ${CMAKE_DL_LIBS})
add_executable(site-test tests/site.c)
target_compile_options(site-test PRIVATE -fno-builtin -fno-optimize-sibling-calls)
target_link_libraries(site-test PRIVATE ${CMAKE_DL_LIBS})
add_executable(measure bench/measure.c)
add_executable(lifetimes tests/lifetimes.c)
target_compile_options(lifetimes PRIVATE -fno-builtin)
add_executable(cost tests/cost.c)
target_compile_options(cost PRIVATE -fno-builtin)

add_executable(live-test tests/live.c)
target_compile_options(live-test PRIVATE -fno-builtin -fno-optimize-sibling-calls)
target_link_libraries(live-test PRIVATE Threads::Threads ${CMAKE_DL_LIBS})

if(NOT APPLE)
 add_executable(server-sim fragbench/server-sim.c)
 target_compile_options(server-sim PRIVATE -fno-builtin -fno-optimize-sibling-calls)
 # Identical-code folding would give distinct lifetime wrappers one debug name.
 if(CMAKE_C_COMPILER_ID STREQUAL "GNU")
  target_compile_options(server-sim PRIVATE -fno-ipa-icf)
 endif()
 target_link_libraries(server-sim PRIVATE Threads::Threads)
endif()

add_executable(defects tests/defects.c)
target_compile_options(defects PRIVATE -UNDEBUG -fno-builtin -fno-optimize-sibling-calls)
target_link_libraries(defects PRIVATE Threads::Threads ${CMAKE_DL_LIBS})

add_executable(hybrid-test tests/hybrid.c)
target_compile_options(hybrid-test PRIVATE -UNDEBUG)
